# .env.example — the secrets this service needs.
# `usvc_seller secrets upload .env.example` seeds the values (for testing) and
# the descriptions (shown to customers who must supply the same names,
# unitysvc/unitysvc#1618). Values use ${NAME:-default}: the real value comes
# from the environment (a shell export, or a GitHub Actions secret/variable in
# CI), else the default. Also sourceable for local runs: `set -a; . ./.env.example`.

# Your upstream HTTP base URL, e.g. https://api.example.com
HTTP_RELAY_BASE_URL=${HTTP_RELAY_BASE_URL:-https://echo.unitysvc.dev}

# Upstream API key — the value selects how the gateway authenticates to your
# upstream:
#   (empty)     strip svcpass; authenticate via your own header instead
#   <literal>   inject this key on the same header your client used
#   __forward__ pass svcpass through (trusted internal hosts only)
HTTP_RELAY_API_KEY=${HTTP_RELAY_API_KEY:-}

# SMTP hostname http-to-smtp delivers through, e.g. smtp.gmail.com,
# smtp.sendgrid.net, or email-smtp.us-east-1.amazonaws.com.
HTTP_TO_SMTP_HOST=${HTTP_TO_SMTP_HOST:-mailpit.unitysvc.dev}

# SMTP port http-to-smtp connects on. 587 for STARTTLS, 465 for implicit TLS,
# 1025 for a plaintext testing sink.
HTTP_TO_SMTP_PORT=${HTTP_TO_SMTP_PORT:-1025}

# Transport security for the http-to-smtp connection: starttls, ssl, or none.
# Use starttls or ssl for anything crossing a public network.
HTTP_TO_SMTP_TLS=${HTTP_TO_SMTP_TLS:-none}

# SMTP username http-to-smtp authenticates with. Usually your sending address or
# a provider API username; leave empty for unauthenticated sinks.
HTTP_TO_SMTP_USERNAME=${HTTP_TO_SMTP_USERNAME:-}

# SMTP password http-to-smtp authenticates with. Leave empty for unauthenticated
# sinks.
HTTP_TO_SMTP_PASSWORD=${HTTP_TO_SMTP_PASSWORD:-}

# Envelope sender http-to-smtp falls back to when the posted payload carries no
# "from". Most SMTP servers require an address they are willing to send as.
HTTP_TO_SMTP_FROM=${HTTP_TO_SMTP_FROM:-http-to-smtp@mailpit.unitysvc.dev}

# Recipient http-to-smtp falls back to when the posted payload carries no "to".
HTTP_TO_SMTP_TO=${HTTP_TO_SMTP_TO:-ops-test@mailpit.unitysvc.dev}

# Route on the UnitySVC SMTP Gateway that http-to-smtp-gateway delivers to, used
# as the SMTP username — a service route (smtp-byok, smtp-to-mailbox), an alias
# (a/alerts), a broadcast group (b/relay), or an enrollment code.
HTTP_TO_SMTP_GATEWAY_TARGET=${HTTP_TO_SMTP_GATEWAY_TARGET:-smtp-byok}

# Envelope sender http-to-smtp-gateway falls back to when the posted payload
# carries no "from".
HTTP_TO_SMTP_GATEWAY_FROM=${HTTP_TO_SMTP_GATEWAY_FROM:-http-to-smtp-gateway@mailpit.unitysvc.dev}

# Recipient http-to-smtp-gateway falls back to when the posted payload carries no
# "to". SMTP Gateway routes resolve on the username, but a recipient still
# travels with the message for any leg that really relays it.
HTTP_TO_SMTP_GATEWAY_TO=${HTTP_TO_SMTP_GATEWAY_TO:-ops-test@mailpit.unitysvc.dev}
